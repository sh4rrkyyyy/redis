cmake_minimum_required(VERSION 3.25)
project(redis-standalone CXX)

set(CMAKE_CXX_STANDARD             23)
set(CMAKE_MODULE_PATH              "${CMAKE_SOURCE_DIR}/cmake")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
set(CMAKE_EXPORT_COMPILE_COMMANDS  ON)

find_package(Catch REQUIRED)
find_package(TBB   REQUIRED)

include(cmake/TestSolution.cmake)
include(cmake/BuildFlags.cmake)
include_directories(utils)

function(add_if_exists name)
  if (EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/${name})
    add_subdirectory(${name})
  endif()
endfunction()

if (UNIX)
  include(cmake/Protobuf.cmake)
  set(Unwind_LIBRARY   "/usr/lib/llvm-20/lib/libunwind.so")
  set(Unwind_INCLUDE_DIR "/usr/lib/llvm-20/include")
  find_package(glog)

  if (Protobuf_FOUND AND glog_FOUND)
    add_if_exists(cactus)
    add_if_exists(io)
    add_if_exists(coro)
    add_if_exists(net)
    add_if_exists(query-builder)
    add_if_exists(query-parser)
    add_if_exists(redis)
  else()
    message(WARNING "redis disabled. Protobuf:${Protobuf_FOUND} glog:${glog_FOUND}")
  endif()
else()
  message(WARNING "Networking tasks support only Linux")
endif()
